 463//     //   the scheduler may have been requested to stop while waiting
 464//     //   for owned inputs to become available. This leads to undefined
 465//     //   behavior if we proceed to execute nodes after stop_ was set.
 466//     nvtxRangePush("Acquire owned inputs");

 479//     //

 481//     //   can execute them in order. The topological order also takes into account
 482//     //   in-place operations, so inputs or siblings is not changed before they are
 483//     //   executed.
 484//     //

 486//     //   as async consumers from the next section may depend on work done on this stream,
 487//     //   and we have no guarantee that the async producer at the end of this section

 528void Scheduler::acquire_owned_inputs(GraphPlan::vertex_descriptor v) {
 529  const auto  idx        = boost::get(boost::vertex_index, graph_, v);
 530  const auto &np         = graph_[v];
 531  auto       &nrt        = node_rts_.at(idx);
 532  const auto &owned_mask = np.infer.owned_inputs;
 533  auto       *task       = std::holds_alternative<SyncRt>(nrt)
 534                               ? static_cast<core::ITask *>(std::get<SyncRt>(nrt).task)
 535                               : static_cast<core::ITask *>(std::get<AsyncRt>(nrt).task);
 536
 537  for (size_t i = 0; i < owned_mask.size(); i++) {
 538    if (!owned_mask.at(i))
 539      continue;
 540
 541    std::optional<core::TView> tview = task->acquire_input(static_cast<int>(i));
 542    while (!tview.has_value()) {
 543      if (stop_.load())
 544        return;
 545      tview = task->acquire_input(static_cast<int>(i));
 546    }
 547
 548    // tviews_.at(np.in_tids.at(i)) = tview.value();
 549  }

 622void Scheduler::run_sync(GraphPlan::vertex_descriptor v) {
 623  using clock     = std::chrono::high_resolution_clock;
 624  const auto  idx = boost::get(boost::vertex_index, graph_, v);
 625  const auto &np  = graph_[v];
 626  auto       &nrt = node_rts_.at(idx);
 627  HOLOFLOW_CHECK(std::holds_alternative<SyncRt>(nrt));
 628  auto &srt = std::get<SyncRt>(nrt);
 629
 630  logger()->trace("[Scheduler::run_sync] Executing node '{}'", np.spec.name);
 631  const auto range_name = task_range_name("sync execute", np.spec);
 632  auto       t0         = clock::now();
 633  auto       r          = core::OpResult::Cancelled;
 634  {
 635    nvtx3::scoped_range task_range{range_name.c_str()};
 636    r = srt.task->execute(srt.ctx);
 637  }
 638  auto t1 = clock::now();
 639
 640  switch (r) {
 641  case core::OpResult::Cancelled:
 642    logger()->debug("[Scheduler::run_sync] Node '{}' execution cancelled", np.spec.name);
 643    stop_.store(true);
 644    break;
 645  case core::OpResult::Eof:
 646    logger()->debug("[Scheduler::run_sync] Node '{}' reached end of stream", np.spec.name);
 647    stop_.store(true);
 648    break;
 649  case core::OpResult::NotReady:
 650    logger()->error(
 651        "[Scheduler::run_sync] The synchronous task '{}' returned NotReady, which is not allowed",
 652        np.spec.name);
 653    stop_.store(true);
 654    break;
 655  case core::OpResult::Ok:
 656    // All good.
 657    break;
 658  }
 659
 660  if (r == core::OpResult::Ok) {
 661    const auto duration_ns = std::chrono::duration_cast<std::chrono::nanoseconds>(t1 - t0).count();
 662    auto [host_in, device_in] =
 663        sum_bytes(std::span<const core::TView>(srt.in_views.data(), srt.in_views.size()));
 664    auto [host_out, device_out] =
 665        sum_bytes(std::span<const core::TView>(srt.out_views.data(), srt.out_views.size()));
 666    record_node_sample(idx, static_cast<uint64_t>(duration_ns), host_in + host_out,
 667                       device_in + device_out);
 668  }
 669}

 673  const auto  idx = boost::get(boost::vertex_index, graph_, v);

 702    break;

 712    const auto duration_ns = std::chrono::duration_cast<std::chrono::nanoseconds>(t1 - t0).count();
 713    auto [host_out, device_out] =
 714        sum_bytes(std::span<const core::TView>(art.out_views.data(), art.out_views.size()));
 715    record_node_sample(idx, static_cast<uint64_t>(duration_ns), host_out, device_out);

 807void Scheduler::start_metrics_thread() {
 808  if (metrics_interval_.count() <= 0) {
 809    return;
 810  }

 812  if (!metrics_running_.compare_exchange_strong(expected, true)) {
 813    return;

 818void Scheduler::stop_metrics_thread() {
 819  bool expected = true;
 820  if (!metrics_running_.compare_exchange_strong(expected, false)) {
 821    return;
 822  }
 823  metrics_cv_.notify_all();
 824  if (metrics_thread_.joinable()) {
 825    metrics_thread_.join();
 826  }
 827  metrics_thread_ = std::thread();
 828}
 829
 830void Scheduler::metrics_loop() {
 831  auto                         last = std::chrono::steady_clock::now();
 832  std::unique_lock<std::mutex> lock(metrics_thread_mutex_);
 833  while (metrics_running_.load()) {
 834    auto interval = metrics_interval_;
 835    if (metrics_cv_.wait_for(lock, interval, [this] { return !metrics_running_.load(); })) {
 836      break;
 837    }
 838    auto now     = std::chrono::steady_clock::now();
 839    auto elapsed = std::chrono::duration<double>(now - last).count();
 840    last         = now;
 841    aggregate_metrics(elapsed);
 842  }
 843  auto now     = std::chrono::steady_clock::now();
 844  auto elapsed = std::chrono::duration<double>(now - last).count();
 845  aggregate_metrics(elapsed);
 846}

 848void Scheduler::aggregate_metrics(double interval_seconds) {
 849  if (interval_seconds <= 0.0) {
 850    interval_seconds = static_cast<double>(metrics_interval_.count()) / 1000.0;
 851    if (interval_seconds <= 0.0) {
 852      interval_seconds = 1.0;
 853    }
 854  }
 855
 856  std::map<std::string, NodeMetrics> snapshot;
 857
 858  for (std::size_t idx = 0; idx < metric_accumulators_.size(); ++idx) {
 859    auto      &acc          = metric_accumulators_.at(idx);
 860    const auto duration_ns  = acc.duration_ns.exchange(0, std::memory_order_relaxed);
 861    const auto runs         = acc.run_count.exchange(0, std::memory_order_relaxed);
 862    const auto host_bytes   = acc.host_bytes.exchange(0, std::memory_order_relaxed);
 863    const auto device_bytes = acc.device_bytes.exchange(0, std::memory_order_relaxed);
 864
 865    NodeMetrics metrics;
 866    metrics.sample_count = runs;
 867    if (runs > 0) {
 868      metrics.average_duration_ms =
 869          static_cast<double>(duration_ns) / static_cast<double>(runs) / 1'000'000.0;
 870    }
 871    if (interval_seconds > 0.0) {
 872      metrics.runs_per_second                  = static_cast<double>(runs) / interval_seconds;
 873      metrics.host_throughput_bytes_per_second = static_cast<double>(host_bytes) / interval_seconds;
 874      metrics.device_throughput_bytes_per_second =
 875          static_cast<double>(device_bytes) / interval_seconds;
 876    }
 877    snapshot.emplace(node_names_.at(idx), metrics);
 878  }
 879
 880  std::lock_guard<std::mutex> lock(metrics_mutex_);
 881  latest_metrics_ = std::move(snapshot);
 882}

 884void Scheduler::record_node_sample(std::size_t idx, uint64_t duration_ns, uint64_t host_bytes,
 885                                   uint64_t device_bytes) {
 886  if (idx >= metric_accumulators_.size()) {
 887    return;
 888  }
 889  auto &acc = metric_accumulators_.at(idx);
 890  acc.duration_ns.fetch_add(duration_ns, std::memory_order_relaxed);
 891  acc.run_count.fetch_add(1, std::memory_order_relaxed);
 892  acc.host_bytes.fetch_add(host_bytes, std::memory_order_relaxed);
 893  acc.device_bytes.fetch_add(device_bytes, std::memory_order_relaxed);
 894}
