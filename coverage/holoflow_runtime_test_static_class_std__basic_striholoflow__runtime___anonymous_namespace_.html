 289                                                    std::unique_ptr<CompilerOutput> prev) {

 339  }
 340
 341  // Stop the total compilation timer safely
 342  total_trace.reset();
 343
 344  if (config_.enable_profiling) {
 345    // profiler_.log_summary(logger_);
 346    run_pass("Dump log summary", [&] { profiler_.log_summary(logger_); });
 347    if (!config_.log_dir.empty()) {
 348      // profiler_.dump_chrome_tracing(config_.log_dir / config_.trace_filename);
 349      run_pass("Dump Chrome Tracing",
 350               [&] { profiler_.dump_chrome_tracing(config_.log_dir / config_.trace_filename); });
 351    }
 352  }
 353
 354  return std::move(out_);
 355}

 377// -------------------------------------------------------------------------------------------------
 378// Pass: Validate Spec
 379// -------------------------------------------------------------------------------------------------
 380void Compiler::Impl::validate_spec() {
 381  std::unordered_set<std::string> names;
 382  std::unordered_set<std::string> edge_dsts;
 383
 384  auto vertices = boost::make_iterator_range(boost::vertices(*gspec_));
 385  for (const auto &v : vertices) {
 386    const auto &ns = (*gspec_)[v];
 387    if (!names.insert(ns.name).second) {
 388      throw CompilerException(std::format("Duplicate node name: '{}'", ns.name));
 389    }
 390    if (!registry_.is_registered(ns.kind)) {
 391      throw CompilerException(std::format("Unknown node kind '{}'", ns.kind));
 392    }
 393  }
 394
 395  auto edges = boost::make_iterator_range(boost::edges(*gspec_));
 396  for (const auto &e : edges) {
 397    const auto &es    = (*gspec_)[e];
 398    const auto  dst   = (*gspec_)[boost::target(e, *gspec_)];
 399    std::string label = std::format("{}:{}", dst.name, es.in_idx);
 400
 401    if (!edge_dsts.insert(label).second) {
 402      throw CompilerException(std::format("Multiple edges targeting: {}", label));
 403    }
 404  }
 405}

 410void Compiler::Impl::build_graph_structure() {
 411  using VSpec = core::GraphSpec::vertex_descriptor;
 412  using VPlan = GraphPlan::vertex_descriptor;
 413  std::map<VSpec, VPlan> v_map;
 414  auto                  &g = out_->graph;
 415
 416  for (auto v : boost::make_iterator_range(boost::vertices(*gspec_))) {

 418    np.spec  = (*gspec_)[v];

 420  }

 423    const auto &es  = (*gspec_)[e];
 424    const auto  src = v_map.at(boost::source(e, *gspec_));
 425    const auto  dst = v_map.at(boost::target(e, *gspec_));
 426    EdgePlan    ep;
 427    ep.spec = es;
 428    boost::add_edge(src, dst, ep, g);
 429  }

 432// -------------------------------------------------------------------------------------------------
 433// Pass: Type Inference
 434// -------------------------------------------------------------------------------------------------
 435void Compiler::Impl::run_type_inference() {
 436  auto                                     &g = out_->graph;
 437  std::vector<GraphPlan::vertex_descriptor> topo_order;
 438
 439  try {
 440    boost::topological_sort(g, std::back_inserter(topo_order));
 441  } catch (const boost::not_a_dag &) {
 442    throw CompilerException("Graph contains a cycle (loop), which is not allowed.");
 443  }
 444
 445  for (auto v : std::views::reverse(topo_order)) {
 446    auto &node       = g[v];
 447    auto  node_trace = trace_scope(std::format("Infer: {}", node.spec.name), "detail");
 448
 449    auto                     in_degree = boost::in_degree(v, g);
 450    std::vector<core::TDesc> input_descs(in_degree);
 451
 452    for (auto e : boost::make_iterator_range(boost::in_edges(v, g))) {
 453      const auto &edge_plan = g[e];
 454      if (edge_plan.spec.in_idx >= input_descs.size()) {
 455        throw CompilerException("Input index out of bounds");
 456      }
 457      input_descs[edge_plan.spec.in_idx] = edge_plan.desc;
 458    }
 459
 460    const auto &factory = registry_.get(node.spec.kind);
 461    node.infer          = factory.infer(input_descs, node.spec.settings);
 462
 463    for (auto e : boost::make_iterator_range(boost::out_edges(v, g))) {
 464      auto &edge_plan = g[e];
 465      if (edge_plan.spec.out_idx >= node.infer.output_descs.size()) {
 466        throw CompilerException("Output index out of bounds");
 467      }
 468      edge_plan.desc = node.infer.output_descs[edge_plan.spec.out_idx];
 469    }
 470  }
 471}

 481  std::vector<GraphPlan::vertex_descriptor> topo;
 482  boost::topological_sort(g, std::back_inserter(topo));
 483
 484  for (auto v : std::views::reverse(topo)) {
 485    auto &node = g[v];
 486
 487    node.in_tids.resize(node.infer.input_descs.size());
 488    for (auto e : boost::make_iterator_range(boost::in_edges(v, g))) {
 489      const auto &ep               = g[e];
 490      node.in_tids[ep.spec.in_idx] = ep.tid;
 491      res.tensor_descs[ep.tid]     = ep.desc;
 492    }
 493
 494    node.out_tids.resize(node.infer.output_descs.size());
 495    auto out_edges = boost::out_edges(v, g);
 496
 497    for (size_t i = 0; i < node.out_tids.size(); ++i) {
 498      int tid               = next_tid++;
 499      node.out_tids[i]      = tid;
 500      res.tensor_descs[tid] = node.infer.output_descs[i];
 501
 502      for (auto e : boost::make_iterator_range(out_edges)) {
 503        if (g[e].spec.out_idx == static_cast<int>(i)) {
 504          g[e].tid = tid;
 505        }
 506      }
 507    }
 508  }
 509}

 516  auto &res = out_->resources;
 517

 519  int next_sid = 0;
 520
 521  std::vector<GraphPlan::vertex_descriptor> topo;
 522  boost::topological_sort(g, std::back_inserter(topo));
 523
 524  for (auto v : std::views::reverse(topo)) {
 525    auto &node = g[v];
 526

 528      int out_tid = node.out_tids[out_idx];
 529      int sid     = -1;
 530
 531      for (const auto &ip : node.infer.in_place) {
 532        if (ip.out_idx == static_cast<int>(out_idx)) {
 533          int in_tid = node.in_tids[ip.in_idx];
 534
 535          if (res.tid_to_sid.contains(in_tid)) {
 536            sid = (int)res.tid_to_sid.at(in_tid);
 537          } else {
 538            throw CompilerException(
 539                std::format("Node '{}': In-place input TID {} has no Storage ID assigned.",
 540                            node.spec.name, in_tid));
 541          }
 542          break;
 543        }
 544      }
 545
 546      if (sid == -1) {
 547        sid = next_sid++;
 548      }
 549      res.tid_to_sid[out_tid] = sid;
 550    }
 551  }

 661void Compiler::Impl::allocate_buffers() {
 662  auto &g   = out_->graph;
 663  auto &res = out_->resources;
 664
 665  res.memory_blocks.clear();
 666  res.storages.clear();
 667
 668  // 1. Identify user-managed SIDs
 669  std::unordered_set<size_t> user_managed_sids;
 670  for (auto v : boost::make_iterator_range(boost::vertices(g))) {
 671    const auto &node = g[v];
 672    for (size_t i = 0; i < node.out_tids.size(); ++i) {
 673      if (node.infer.owned_outputs[i]) {
 674        user_managed_sids.insert(res.tid_to_sid.at(node.out_tids[i]));
 675      }
 676    }
 677    for (size_t i = 0; i < node.in_tids.size(); ++i) {
 678      if (node.infer.owned_inputs[i]) {
 679        user_managed_sids.insert(res.tid_to_sid.at(node.in_tids[i]));
 680      }
 681    }
 682  }
 683
 684  // 2. Map SID to representative TID
 685  std::map<size_t, size_t> sid_to_rep_tid;
 686  for (const auto &[tid, sid] : res.tid_to_sid) {
 687    if (!sid_to_rep_tid.count(sid)) {
 688      sid_to_rep_tid[sid] = tid;
 689    }
 690  }
 691
 692  // 3. Build a pool of scavengable blocks from prev_
 693  // Key: {MemLoc, size_in_bytes}
 694  std::multimap<std::pair<core::MemLoc, size_t>, MemoryBlock> free_blocks;
 695  if (prev_) {
 696    for (auto &[prev_sid, block] : prev_->resources.memory_blocks) {
 697      free_blocks.emplace(std::make_pair(block.mem_loc, block.size_bytes), std::move(block));
 698    }
 699  }
 700
 701  // 4. Allocate or Scavenge
 702  for (const auto &[sid, tid] : sid_to_rep_tid) {
 703    auto        alloc_scope = trace_scope(std::format("Alloc SID {}", sid), "detail");
 704    const auto &desc        = res.tensor_descs.at(tid);
 705
 706    auto storage     = std::make_unique<core::Storage>();
 707    storage->mem_loc = desc.mem_loc;
 708    storage->bytes   = desc.num_bytes();
 709    storage->ptr     = nullptr;
 710
 711    if (!user_managed_sids.contains(sid)) {
 712      MemoryBlock block;
 713      auto        pool_key = std::make_pair(desc.mem_loc, desc.num_bytes());
 714      auto        it       = free_blocks.find(pool_key);
 715
 716      if (it != free_blocks.end()) {
 717        // We found an exact match! Scavenge it.
 718        logger_->info("Reusing {} bytes for SID {} at {:?} memory", desc.num_bytes(), sid,
 719                      to_string(desc.mem_loc));
 720        block = std::move(it->second);
 721        free_blocks.erase(it);
 722      } else {
 723        // No match found, allocate fresh memory.
 724        block.mem_loc    = desc.mem_loc;
 725        block.size_bytes = desc.num_bytes();
 726
 727        logger_->info("Allocating {} bytes for SID {} at {:?} memory", block.size_bytes, sid,
 728                      to_string(desc.mem_loc));
 729
 730        {
 731          auto sys_scope = trace_scope(
 732              desc.mem_loc == core::MemLoc::Host ? "Host Malloc" : "Device Malloc", "syscall");
 733          if (desc.mem_loc == core::MemLoc::Host) {
 734            block.h_data = curaii::make_unique_host_ptr<std::byte>(block.size_bytes);
 735          } else {
 736            block.d_data = curaii::make_unique_device_ptr<std::byte>(block.size_bytes);
 737          }
 738        }
 739      }
 740
 741      storage->ptr = static_cast<std::byte *>(block.get());
 742      res.memory_blocks.emplace(sid, std::move(block));
 743    } else {
 744      logger_->info("SID {} is user-managed; skipping allocation.", sid);
 745    }
 746
 747    res.storages.emplace(sid, std::move(storage));
 748  }
 749
 750  // Temp test, trigger a 1b cuda memcopy to see how it shows up in the profiler
 751  CUDA_CHECK(cudaDeviceSynchronize());
 752  if (res.memory_blocks.size() >= 2) {
 753    auto &block1 = res.memory_blocks.begin()->second;
 754    auto &block2 = std::next(res.memory_blocks.begin())->second;
 755    if (block1.mem_loc == core::MemLoc::Device && block2.mem_loc == core::MemLoc::Device) {
 756      auto sys_scope = trace_scope("Test Memcpy", "syscall");
 757      CUDA_CHECK(cudaMemcpy(block2.get(), block1.get(), 1, cudaMemcpyDeviceToDevice));
 758      CUDA_CHECK(cudaDeviceSynchronize());
 759    }
 760  }
 761
 762  // Any blocks left inside `free_blocks` will naturally go out of scope here and
 763  // safely deallocate, meaning memory for removed nodes is properly cleaned up.
 764}

 766// -------------------------------------------------------------------------------------------------
 767// Pass: Section Partitioning
 768// -------------------------------------------------------------------------------------------------
 769void Compiler::Impl::partition_sections() {
 770  auto &g         = out_->graph;
 771  auto  num_verts = boost::num_vertices(g);
 772
 773  std::vector<size_t> parent(num_verts);
 774  std::iota(parent.begin(), parent.end(), 0);
 775
 776  auto find = [&](size_t i) {
 777    while (i != parent[i]) {
 778      parent[i] = parent[parent[i]];
 779      i         = parent[i];
 780    }
 781    return i;
 782  };
 783
 784  auto unite = [&](size_t i, size_t j) {
 785    size_t root_i = find(i);
 786    size_t root_j = find(j);
 787    if (root_i != root_j)
 788      parent[root_i] = root_j;
 789  };
 790
 791  for (auto e : boost::make_iterator_range(boost::edges(g))) {
 792    auto u = boost::source(e, g);
 793    auto v = boost::target(e, g);
 794    if (g[u].infer.kind == core::TaskKind::Sync && g[v].infer.kind == core::TaskKind::Sync) {
 795      unite(u, v);
 796    }
 797  }
 798
 799  for (auto v : boost::make_iterator_range(boost::vertices(g))) {
 800    if (g[v].infer.kind == core::TaskKind::Async) {
 801      std::vector<size_t> sync_preds;
 802      for (auto e : boost::make_iterator_range(boost::in_edges(v, g))) {
 803        auto p = boost::source(e, g);
 804        if (g[p].infer.kind == core::TaskKind::Sync)
 805          sync_preds.push_back(p);
 806      }
 807      if (!sync_preds.empty()) {
 808        for (size_t i = 1; i < sync_preds.size(); ++i)
 809          unite(sync_preds[0], sync_preds[i]);
 810      }
 811
 812      std::vector<size_t> sync_succs;
 813      for (auto e : boost::make_iterator_range(boost::out_edges(v, g))) {
 814        auto s = boost::target(e, g);
 815        if (g[s].infer.kind == core::TaskKind::Sync)
 816          sync_succs.push_back(s);
 817      }
 818      if (!sync_succs.empty()) {
 819        for (size_t i = 1; i < sync_succs.size(); ++i)
 820          unite(sync_succs[0], sync_succs[i]);
 821      }
 822    }
 823  }
 824
 825  std::map<size_t, size_t> root_to_section_id;
 826  out_->sections.clear();
 827  int next_sec_id = 0;
 828  node_to_section_map_.clear();
 829
 830  auto get_section_id = [&](size_t v_idx) {
 831    size_t root = find(v_idx);
 832    if (root_to_section_id.find(root) == root_to_section_id.end()) {
 833      Section s;
 834      s.id   = next_sec_id;
 835      s.name = std::format("section-{}", next_sec_id);
 836      out_->sections.push_back(s);
 837      root_to_section_id[root] = next_sec_id++;
 838    }
 839    return root_to_section_id[root];
 840  };
 841
 842  std::vector<GraphPlan::vertex_descriptor> topo;
 843  boost::topological_sort(g, std::back_inserter(topo));
 844
 845  for (auto v : std::views::reverse(topo)) {
 846    auto &np = g[v];
 847    if (np.infer.kind == core::TaskKind::Sync) {
 848      size_t sec_id = get_section_id(v);
 849      out_->sections[sec_id].sync_topo.push_back(v);
 850      node_to_section_map_[np.spec.name] = sec_id;
 851    }
 852  }
 853
 854  for (auto v : boost::make_iterator_range(boost::vertices(g))) {
 855    if (g[v].infer.kind != core::TaskKind::Async)
 856      continue;
 857
 858    std::set<size_t> unique_cons_sections;
 859    for (auto e : boost::make_iterator_range(boost::out_edges(v, g))) {
 860      auto s = boost::target(e, g);
 861      if (g[s].infer.kind == core::TaskKind::Sync) {
 862        unique_cons_sections.insert(get_section_id(s));
 863      }
 864    }
 865    for (size_t sec_id : unique_cons_sections) {
 866      out_->sections[sec_id].async_cons.push_back(v);
 867    }
 868
 869    std::set<size_t> unique_prod_sections;
 870    for (auto e : boost::make_iterator_range(boost::in_edges(v, g))) {
 871      auto p = boost::source(e, g);
 872      if (g[p].infer.kind == core::TaskKind::Sync) {
 873        unique_prod_sections.insert(get_section_id(p));
 874      }
 875    }
 876    for (size_t sec_id : unique_prod_sections) {
 877      out_->sections[sec_id].async_prod.push_back(v);
 878    }
 879  }
 880}

 891void Compiler::Impl::assign_streams() {
 892  out_->resources.streams.clear();
 893
 894  // 1. Only set up the iterators if prev_ actually exists
 895  if (prev_) {
 896    auto prev_it  = prev_->resources.streams.begin();
 897    auto prev_end = prev_->resources.streams.end();
 898
 899    for (auto &sec : out_->sections) {
 900      if (prev_it != prev_end) {
 901        // Scavenge an existing stream
 902        auto &old_stream = prev_it->second;
 903        sec.stream       = old_stream.get();
 904
 905        out_->resources.streams.emplace(sec.id, std::move(old_stream));
 906        ++prev_it;
 907      } else {
 908        // Fallback: Create a new stream (ran out of old ones)
 909        curaii::CudaStream stream;
 910        sec.stream = stream.get();
 911        out_->resources.streams.emplace(sec.id, std::move(stream));
 912      }
 913    }
 914  } else {
 915    // 2. No previous graph at all, just create fresh streams for everything
 916    for (auto &sec : out_->sections) {
